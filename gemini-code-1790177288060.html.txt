        async function sendMessage() {
            const input = document.getElementById('userInput');
            const text = input.value.trim();
            if(!text) return;

            const chatBox = document.getElementById('chatBox');
            
            // Render Chat User
            chatBox.innerHTML += `
                <div class="chat-bubble-user p-3.5 max-w-[85%] self-end ml-auto text-white shadow-sm">
                    <p>${escapeHtml(text)}</p>
                </div>
            `;
            input.value = '';
            chatBox.scrollTop = chatBox.scrollHeight;

            // Tampilkan Loading AI
            document.getElementById('analysisBox').classList.remove('hidden');
            document.getElementById('feedbackText').innerText = "Sedang menganalisis gaya komunikasimu...";

            if (geminiApiKey) {
                await analyzeWithGemini(text);
            } else {
                analyzeWithEnhancedHeuristic(text);
            }
        }

        // Analisis Menggunakan Gemini 2.5 Flash
        async function analyzeWithGemini(userText) {
            const sc = scenarios[currentScenarioKey];
            const prompt = `
            Kamu adalah Guru BK & Expert Komunikasi Asertif Gen Z.
            Skenario: ${sc.desc}
            Pesan Lawan Bicara: "${sc.initialMsg}"
            Respon Siswa: "${userText}"

            Tugasmu: Analisis respon siswa apakah Pasif, Agresif, Pasif-Agresif, atau Asertif.
            Kembalikan jawaban HANYA dalam format JSON valid tanpa tanda markdown (```json):
            {
                "category": "ASERTIF" / "PASIF" / "AGRESIF" / "PASIF-AGRESIF",
                "assertiveScore": 0-100,
                "empathyScore": 0-100,
                "feedback": "Penjelasan singkat gaya santai Gen Z kenapa responnya dinilai begitu.",
                "upgradeFormula": "Contoh kalimat asertif yang pas pakai formula I-Message (Saya merasa... kalau... karena... aku mau...)",
                "npcReply": "Balasan balasan santai dari lawan bicara merespon pesan siswa tersebut."
            }`;

            try {
                const response = await fetch(`[https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=$](https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=$){geminiApiKey}`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        contents: [{ parts: [{ text: prompt }] }]
                    })
                });

                const data = await response.json();
                let rawText = data.candidates[0].content.parts[0].text;
                rawText = rawText.replace(/```json/g, '').replace(/```/g, '').trim();
                const res = JSON.parse(rawText);

                renderAnalysisResults(res);
            } catch (e) {
                console.error(e);
                analyzeWithEnhancedHeuristic(userText); // Fallback kalau API error
            }
        }

        // Analisis Heuristik Kasual yang Ditingkatkan
        function analyzeWithEnhancedHeuristic(userText) {
            const textLower = userText.toLowerCase();
            let category = "PASIF";
            let assertiveScore = 40;
            let empathyScore = 50;
            let feedback = "";
            let upgradeFormula = "";
            let npcReply = "";

            // Kasus Agresif / Pasif-Agresif (misal: "dih balikin dulu", "bacot", kata kasar, emosi berlebih)
            if (textLower.startsWith("dih") || textLower.includes("bacot") || textLower.includes("goblok") || textLower.includes("lu aja") || textLower.includes("ga usah") || textLower.includes("apaan si")) {
                category = "AGRESIF / DEFENSIF";
                assertiveScore = 30;
                empathyScore = 20;
                feedback = "Respon kamu terkesan ketus/defensif ('dih...'). Walau kamu berusaha menolak, nada begini bikin lawan bicara defensif dan memicu debat kusir bukannya menyelesaikan masalah.";
                upgradeFormula = "Gunakan alasan tegas tanpa menyindir: 'Gua ga bisa minjemin lagi bro, utang kemarin kan belum beres. Gua butuh uangnya juga buat keperluan lain.'";
                npcReply = "Yah elah santai kali, biasa aja dong ga usah pake 'dih' segala. Gua kan cuma nanya!";
            } 
            // Kasus Pasif
            else if (textLower.includes("yaudah") || textLower.includes("terserah") || textLower.includes("maaf ya") || textLower.length < 5) {
                category = "PASIF";
                assertiveScore = 20;
                empathyScore = 70;
                feedback = "Kamu terlalu mengalah dan enggan menetapkan batas (boundary). Ini bisa bikin kamu dimanfaatkan terus-terusan.";
                upgradeFormula = "Jelaskan posisi kamu secara jujur: 'Sori banget belum bisa, uangnya lagi mau aku pake buat kebutuhan penting minggu ini.'";
                npcReply = "Nah gitu dong! Sip bro, tar malam gua jemput ya!";
            } 
            // Kasus Asertif Bagus
            else {
                category = "ASERTIF";
                assertiveScore = 85;
                empathyScore = 80;
                feedback = "Keren! Komunikasi kamu cukup jelas, langsung ke inti masalah, dan tetap menghargai tanpa emosi berlebihan.";
                upgradeFormula = "Respon kamu udah mantap! Format I-Message: 'Aku mengerti kamu butuh, tapi aku belum bisa minjemin lagi sebelum yang kemarin dilunasin.'";
                npcReply = "Oh gitu ya... Yaudah deh bro, sori ya. Nanti gua usahain pelan-pelan kelarin utang kemarin.";
            }

            renderAnalysisResults({ category, assertiveScore, empathyScore, feedback, upgradeFormula, npcReply });
        }

        function renderAnalysisResults(res) {
            document.getElementById('scoreAssertive').innerText = res.assertiveScore + "%";
            document.getElementById('barAssertive').style.width = res.assertiveScore + "%";
            
            document.getElementById('scoreEmpathy').innerText = res.empathyScore + "%";
            document.getElementById('barEmpathy').style.width = res.empathyScore + "%";

            document.getElementById('badgeCategory').innerText = res.category;
            document.getElementById('feedbackText').innerText = res.feedback;
            document.getElementById('upgradeText').innerText = res.upgradeFormula;

            // Render Balasan NPC
            const chatBox = document.getElementById('chatBox');
            setTimeout(() => {
                chatBox.innerHTML += `
                    <div class="chat-bubble-npc p-3.5 max-w-[85%] self-start shadow-sm">
                        <p class="text-slate-200">${escapeHtml(res.npcReply)}</p>
                    </div>
                `;
                chatBox.scrollTop = chatBox.scrollHeight;
            }, 600);
        }

        function escapeHtml(text) {
            return text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
        }
